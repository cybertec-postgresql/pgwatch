    40  // Since Collect() now reads a snapshot without clearing the cache,
    41  // the cache should still contain the original data after collect.
    42  func TestWriteAfterCollect(t *testing.T) {
    43  	promw := newTestPrometheusWriter("test")
    44  
    45  	// Write initial data
    46  	msg := metrics.MeasurementEnvelope{
    47  		DBName:     "db1",
    48  		MetricName: "metric1",
    49  		Data: metrics.Measurements{
    50  			{metrics.EpochColumnName: time.Now().UnixNano(), "value": int64(100)},
    51  		},
    52  	}
    53  	require.NoError(t, promw.Write(msg))
    54  
    55  	// Collect reads a snapshot — cache is NOT cleared
    56  	ch := make(chan prometheus.Metric, 100)
    57  	promw.Collect(ch)
    58  	assert.NotEmpty(t, promw.Cache, "cache should still contain data after Collect (snapshot-based)")

    62  	require.NoError(t, promw.Write(msg))
    63

   121  			},

   145  	promw := newTestPrometheusWriter("test")

   147  
   148  	// Two identical rows — same labels, same field name

   152  			MetricName: "metric1",

   154  			Data: metrics.Measurements{

   157  					"tag_host":              "server1",
   158  					"value":                 int64(42),
   159  					"bool_val":              false,
   160  					"extra_field1":          "ignored", // extra fields should not affect identity
   161  				},
   162  				{
   163  					metrics.EpochColumnName: time.Now().UnixNano(),
   164  					"tag_host":              "server1",
   165  					"value":                 int64(99), // different values, same identity
   166  					"bool_val":              true,
   167  					"extra_field1":          "ignored", // extra fields should not affect identity
   168  				},
   169  			},
   170  		},
   171  	}
   172  
   173  	ch := make(chan prometheus.Metric, 100)
   174  	promw.Collect(ch)
   175  	close(ch)
   176  
   177  	var count int
   178  	for c := range ch {
   179  		t.Log(c.Desc())
   180  		count++
   181  	}
   182  	// Should deduplicate — only the first occurrence is emitted.
   183  	// 2 data metrics + 3 meta-metrics = 5 total.

   185  }
