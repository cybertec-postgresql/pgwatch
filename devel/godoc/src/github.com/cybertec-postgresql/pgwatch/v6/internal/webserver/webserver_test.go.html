   116  
   117  	tempFile := path.Join(t.TempDir(), "file.ext")
   118  	assert.NoError(t, os.WriteFile(tempFile, []byte("content"), 0644))
   119  	uiFS = mockFS{
   120  		OpenFunc: func(name string) (fs.File, error) {
   121  			switch name {
   122  			case "static/app.js", "assets/index-abc123.js", "favicon.ico":
   123  				return os.Open(tempFile)
   124  			default:
   125  				return nil, fs.ErrNotExist
   126  			}
   127  		},
   128  	}
   129  
   130  	const (
   131  		revalidate = "no-cache"
   132  		immutable  = "public, max-age=31536000, immutable"
   133  	)
   134  	tests := []struct {
   135  		name     string
   136  		basePath string
   137  		url      string
   138  		want     string
   139  	}{
   140  		{"SPA route", "/", "/sources", revalidate},
   141  		{"root", "/", "/", revalidate},
   142  		{"static file", "/", "/static/app.js", immutable},
   143  		{"assets file", "/", "/assets/index-abc123.js", immutable},
   144  		{"other root file", "/", "/favicon.ico", revalidate},
   145  		{"SPA route with base path", "/pgwatch/", "/pgwatch/metrics", revalidate},
   146  		{"assets file with base path", "/pgwatch/", "/pgwatch/assets/index-abc123.js", immutable},
   147  		{"other root file with base path", "/pgwatch/", "/pgwatch/favicon.ico", revalidate},
   148  	}
   149  	for _, tt := range tests {
   150  		t.Run(tt.name, func(t *testing.T) {
   151  			ts := &WebUIServer{
   152  				Logger:    logrus.StandardLogger(),
   153  				basePath:  tt.basePath,
   154  				indexHTML: []byte("<html></html>"),
   155  			}
   156  			w := httptest.NewRecorder()
   157  			ts.handleStatic(w, httptest.NewRequest(http.MethodGet, tt.url, nil))
   158  			resp := w.Result()
   159  			defer resp.Body.Close()
   160  			assert.Equal(t, http.StatusOK, resp.StatusCode)
   161  			assert.Equal(t, tt.want, resp.Header.Get("Cache-Control"))
   162  		})
   163  	}
