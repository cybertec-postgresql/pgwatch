    22  // writeTestCert writes a self-signed certificate and its key as PEM files
    23  // into dir and returns their paths.
    24  func writeTestCert(t *testing.T, dir string) (certFile, keyFile string) {
    25  	t.Helper()
    26  	key, err := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
    27  	require.NoError(t, err)
    28  	tmpl := &x509.Certificate{
    29  		SerialNumber:          big.NewInt(1),
    30  		Subject:               pkix.Name{CommonName: "pgwatch-test"},
    31  		NotBefore:             time.Now().Add(-time.Hour),
    32  		NotAfter:              time.Now().Add(time.Hour),
    33  		IsCA:                  true,
    34  		BasicConstraintsValid: true,
    35  	}
    36  	der, err := x509.CreateCertificate(rand.Reader, tmpl, tmpl, &key.PublicKey, key)
    37  	require.NoError(t, err)
    38  	keyDer, err := x509.MarshalECPrivateKey(key)
    39  	require.NoError(t, err)
    40  
    41  	certFile = filepath.Join(dir, "cert.pem")
    42  	keyFile = filepath.Join(dir, "key.pem")
    43  	require.NoError(t, os.WriteFile(certFile, pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: der}), 0o600))
    44  	require.NoError(t, os.WriteFile(keyFile, pem.EncodeToMemory(&pem.Block{Type: "EC PRIVATE KEY", Bytes: keyDer}), 0o600))
    45  	return certFile, keyFile
    46  }
