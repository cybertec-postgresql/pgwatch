   284  // TestBlockingPool covers the wedged-pool fault injector in mocks.go.
   285  // Every BlockingPool method blocks until ctx.Done(); pre-cancelling the
   286  // context makes each call return immediately with ctx.Err() so the
   287  // assertions are deterministic and finish in microseconds.
   288  func TestBlockingPool(t *testing.T) {
   289  	pool := testutil.BlockingPool{}
   290  
   291  	cancelledCtx := func() context.Context {
   292  		ctx, cancel := context.WithCancel(context.Background())
   293  		cancel()
   294  		return ctx
   295  	}
   296  
   297  	t.Run("Ping", func(t *testing.T) {
   298  		assert.ErrorIs(t, pool.Ping(cancelledCtx()), context.Canceled)
   299  	})
   300  
   301  	t.Run("Query", func(t *testing.T) {
   302  		rows, err := pool.Query(cancelledCtx(), "select 1")
   303  		assert.ErrorIs(t, err, context.Canceled)
   304  		assert.Nil(t, rows)
   305  	})
   306  
   307  	t.Run("QueryRow", func(t *testing.T) {
   308  		var dst any
   309  		assert.ErrorIs(t, pool.QueryRow(cancelledCtx(), "select 1").Scan(&dst), context.Canceled)
   310  	})
   311  
   312  	t.Run("Exec", func(t *testing.T) {
   313  		tag, err := pool.Exec(cancelledCtx(), "x")
   314  		assert.ErrorIs(t, err, context.Canceled)
   315  		assert.Empty(t, tag)
   316  	})
   317  
   318  	t.Run("Acquire", func(t *testing.T) {
   319  		conn, err := pool.Acquire(cancelledCtx())
   320  		assert.ErrorIs(t, err, context.Canceled)
   321  		assert.Nil(t, conn)
   322  	})
   323  
   324  	t.Run("Close", func(t *testing.T) {
   325  		assert.NotPanics(t, func() { pool.Close() })
   326  	})
   327  
   328  	t.Run("SendBatch", func(t *testing.T) {
   329  		br := pool.SendBatch(cancelledCtx(), &pgx.Batch{}).(*testutil.BlockingBatchResults)
   330  
   331  		_, err := br.Query()
   332  		assert.ErrorIs(t, err, context.Canceled)
   333  
   334  		_, err = br.Exec()
   335  		assert.ErrorIs(t, err, context.Canceled)
   336  
   337  		var dst any
   338  		assert.ErrorIs(t, br.QueryRow().Scan(&dst), context.Canceled)
   339  
   340  		assert.NoError(t, br.Err())                       // before Close
   341  		assert.NoError(t, br.Close())
   342  		assert.ErrorIs(t, br.Err(), context.Canceled)     // after Close
   343  	})
   344  }
