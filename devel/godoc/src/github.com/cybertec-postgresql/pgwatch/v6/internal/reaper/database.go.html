    81  // GCDSlice computes GCD across a slice. Returns 0 for empty input.
    82  func GCDSlice(vals []int) int {
    83  	if len(vals) == 0 {
    84  		return 0
    85  	}
    86  	g := vals[0]
    87  	for _, v := range vals[1:] {
    88  		for v != 0 {
    89  			g, v = v, g%v
    90  		}
    91  	}
    92  	return g
    93  }

    95  // calcTickInterval computes GCD of all metric intervals with a minimum floor.
    96  func (sr *DbConnReaper) calcTickInterval() time.Duration {
    97  	am := sr.activeMetrics()
    98  	intervals := make([]int, 0, len(am))
    99  	for _, d := range am {
   100  		intervals = append(intervals, max(int(d.Seconds()), minTickInterval))
   101  	}
   102  	return time.Duration(max(GCDSlice(intervals), minTickInterval)) * time.Second
   103  }

   154  // Run is the main loop for a single source. It replaces N per-metric goroutines
   155  // with one goroutine that batches SQL queries at GCD-aligned ticks.
   156  func (sr *DbConnReaper) Reap(ctx context.Context) {
   157  	l := log.GetLogger(ctx).WithField("source", sr.md.Name)
   158  	ctx = log.WithLogger(ctx, l)
   159  	var err error
   160  	for {
   161  		if err = sr.md.FetchRuntimeInfo(ctx, false); err != nil {
   162  			l.WithError(err).Warning("could not refresh runtime info")
   163  		}
   164  
   165  		now := time.Now()
   166  		var batch []batchEntry
   167  
   168  		for name, interval := range sr.activeMetrics() {
   169  			if interval <= 0 {
   170  				continue
   171  			}
   172  			if lf := sr.lastFetch[name]; !lf.IsZero() && now.Sub(lf) < interval {
   173  				continue
   174  			}
   175  
   176  			metric, ok := metricDefs.GetMetricDef(name)
   177  			if !ok || sr.isRoleExcluded(metric) {
   178  				continue
   179  			}
   180  			switch {
   181  			case name == specialMetricServerLogEventCounts:
   182  				if !sr.logParserStarted {
   183  					sr.logParserStarted = true // streaming parser starts once per worker lifetime and runs until ctx cancel
   184  					go func() {
   185  						if e := sr.runLogParser(ctx); e != nil {
   186  							l.WithError(e).Error("log parser error")
   187  						}
   188  					}()
   189  				}
   190  			case IsDirectlyFetchableMetric(sr.md, name):
   191  				err = sr.fetchOSMetric(ctx, name)
   192  				sr.lastFetch[name] = time.Now()
   193  			case name == specialMetricChangeEvents || name == specialMetricInstanceUp:
   194  				err = sr.fetchSpecialMetric(ctx, name, metric.StorageName)
   195  				sr.lastFetch[name] = time.Now()
   196  			default:
   197  				if cached := sr.reaper.GetMeasurementCache(sr.cacheKey(metric, name)); len(cached) > 0 {
   198  					l.WithField("metric", name).Info("instance level cache hit")
   199  					sr.sendEnvelope(ctx, name, metric.StorageName, cached)
   200  					sr.lastFetch[name] = time.Now()
   201  					break
   202  				}
   203  				sr.md.RLock()
   204  				version := sr.md.Version
   205  				sr.md.RUnlock()
   206  				sql := metric.GetSQL(version)
   207  				if sql == "" {
   208  					l.WithField("source", sr.md.Name).WithField("version", version).Warning("no SQL found for metric version")
   209  					sr.lastFetch[name] = time.Now()
   210  					break
   211  				}
   212  				if sr.isDegraded(name) {
   213  					if err = sr.fetchMetric(ctx, batchEntry{metricName: name, metric: metric, sql: sql}); err != nil {
   214  						l.WithError(err).WithField("metric", name).Error("degraded metric fetch failed")
   215  					} else {
   216  						l.WithField("metric", name).Info("degraded metric recovered, returning to batch execution")
   217  						sr.clearDegraded(name)
   218  					}
   219  					sr.lastFetch[name] = time.Now()
   220  					break
   221  				}
   222  				batch = append(batch, batchEntry{metricName: name, metric: metric, sql: sql})
   223  			}
   224  			if err != nil {
   225  				l.WithError(err).WithField("metric", name).Error("failed to fetch metric")
   226  			}
   227  		}
   228  
   229  		if len(batch) > 0 {
   230  			if sr.md.IsPostgresSource() {
   231  				err = sr.executeBatch(ctx, batch)
   232  			} else {
   233  				for _, e := range batch {
   234  					err = sr.fetchMetric(ctx, e)
   235  				}
   236  			}
   237  			if err != nil {
   238  				l.WithError(err).Error("failed to fetch metrics")
   239  			}
   240  
   241  			now := time.Now()
   242  			for _, e := range batch {
   243  				sr.lastFetch[e.metricName] = now
   244  			}
   245  		}
   246  		select {
   247  		case <-ctx.Done():
   248  			return
   249  		case <-time.After(sr.calcTickInterval()):
   250  		}
   251  	}
   252  }

   254  // executeBatch sends all SQLs in a single pgx.Batch round-trip, dispatching
   255  // each result immediately as it arrives. If any query fails, PostgreSQL's
   256  // extended protocol aborts all subsequent queries in the same sync boundary
   257  // (cascade failure). Any entry that returns an error from the batch is retried
   258  // individually via fetchMetric to isolate real failures from cascade failures.
   259  // Entries that fail even after the individual retry are marked as degraded
   260  // so that subsequent runs use fetchMetric for them until they recover.
   261  func (sr *DbConnReaper) executeBatch(ctx context.Context, entries []batchEntry) error {
   262  	batch := &pgx.Batch{}
   263  	for _, e := range entries {
   264  		batch.Queue(e.sql)
   265  	}
   266  
   267  	// Bound the batch round-trip by the tick interval (with the floor from
   268  	// db.MinFetchTimeout) so a wedged connection cannot stall the source
   269  	// worker past the next tick. The derived ctx embeds an op-named cause
   270  	// so the error chain is greppable per call site.
   271  	bctx, cancel := db.WithFetchTimeout(ctx, "batch", sr.calcTickInterval())
   272  	defer cancel()
   273  
   274  	br := sr.md.Conn.SendBatch(bctx, batch)
   275  	defer func() { _ = br.Close() }()
   276  
   277  	var (
   278  		errs    []error
   279  		retries []batchEntry
   280  	)
   281  	for _, e := range entries {
   282  		rows, err := br.Query()
   283  		if err != nil {
   284  			// May be a real error or a cascade from an earlier failure; retry individually.
   285  			retries = append(retries, e)
   286  			continue
   287  		}
   288  		errs = append(errs, sr.CollectAndDispatch(ctx, rows, e.metricName, e.metric))
   289  	}
   290  
   291  	for _, e := range retries {
   292  		if err := sr.fetchMetric(ctx, e); err != nil {
   293  			errs = append(errs, fmt.Errorf("failed to fetch metric %s: %v", e.metricName, err))
   294  			log.GetLogger(ctx).WithField("metric", e.metricName).Warning("metric degraded after repeated failures, switching to individual fetch")
   295  			sr.markDegraded(e.metricName)
   296  		}
   297  	}
   298  	if joined := errors.Join(errs...); joined != nil {
   299  		return fmt.Errorf("batch: %w", joined)
   300  	}
   301  	return nil
   302  }

   304  // fetchMetric executes a single SQL query and returns the resulting measurements.
   305  func (sr *DbConnReaper) fetchMetric(ctx context.Context, entry batchEntry) error {
   306  	// Bound the per-metric round-trip by max(GetMetricInterval, MinFetchTimeout).
   307  	// GetMetricInterval already returns a time.Duration in seconds — do NOT
   308  	// multiply by time.Second again.
   309  	interval := sr.md.GetMetricInterval(entry.metricName)
   310  	op := "fetch " + entry.metricName
   311  	fctx, cancel := db.WithFetchTimeout(ctx, op, interval)
   312  	defer cancel()
   313  	if err := sr.fetchMetricOnce(fctx, entry); err != nil {
   314  		return fmt.Errorf("%s: %w", op, err)
   315  	}
   316  	return nil
   317  }

   412  // QueryMeasurements runs the given SQL on the source connection and returns
   413  // the collected rows. The round-trip is bounded by db.ChangeDetectionTimeout
   414  // so a hung source cannot block the change-detection sweep indefinitely.
   415  func QueryMeasurements(ctx context.Context, md *sources.DbConn, sql string, args ...any) (metrics.Measurements, error) {
   416  	if strings.TrimSpace(sql) == "" {
   417  		return nil, errors.New("empty SQL")
   418  	}
   419  	qctx, cancel := db.WithOpTimeout(ctx, "change_detection_query", db.ChangeDetectionTimeout)
   420  	defer cancel()
   421  	// For non-postgres connections (e.g. pgbouncer, pgpool), use simple protocol
   422  	if !md.IsPostgresSource() {
   423  		args = append([]any{pgx.QueryExecModeSimpleProtocol}, args...)
   424  	}
   425  	// lock_timeout is set at connection level via RuntimeParams, no need for transaction wrapper
   426  	rows, err := md.Conn.Query(qctx, sql, args...)
   427  	if err == nil {
   428  		return pgx.CollectRows(rows, metrics.RowToMeasurement)
   429  	}
   430  	if errors.Is(err, context.DeadlineExceeded) {
   431  		return nil, fmt.Errorf("change_detection_query: %w", context.Cause(qctx))
   432  	}
   433  	return nil, err
   434  }

   436  func (r *reaper) DetectSprocChanges(ctx context.Context, md *sources.DbConn) (changeCounts ChangeDetectionResults) {
   437  	detectedChanges := make(metrics.Measurements, 0)
   438  	var firstRun bool
   439  	l := log.GetLogger(ctx)
   440  	changeCounts.Target = "functions"
   441  	l.Debug("checking for sproc changes...")
   442  	if _, ok := md.ChangeState["sproc_hashes"]; !ok {
   443  		firstRun = true
   444  		md.ChangeState["sproc_hashes"] = make(map[string]string)
   445  	}
   446  	mvp, ok := metricDefs.GetMetricDef("sproc_hashes")
   447  	if !ok {
   448  		l.Error("could not get sproc_hashes sql")
   449  		return
   450  	}
   451  	data, err := QueryMeasurements(ctx, md, mvp.GetSQL(int(md.Version)))
   452  	if err != nil {
   453  		l.Error(err)
   454  		return
   455  	}
   456  	for _, dr := range data {
   457  		objIdent := dr["tag_sproc"].(string) + dbMetricJoinStr + dr["tag_oid"].(string)
   458  		prevHash, ok := md.ChangeState["sproc_hashes"][objIdent]
   459  		ll := l.WithField("sproc", dr["tag_sproc"]).WithField("oid", dr["tag_oid"])
   460  		if ok { // we have existing state
   461  			if prevHash != dr["md5"].(string) {
   462  				ll.Debug("change detected")
   463  				dr["event"] = "alter"
   464  				detectedChanges = append(detectedChanges, dr)
   465  				md.ChangeState["sproc_hashes"][objIdent] = dr["md5"].(string)
   466  				changeCounts.Altered++
   467  			}
   468  		} else { // check for new / delete
   469  			if !firstRun {
   470  				ll.Debug("new sproc detected")
   471  				dr["event"] = "create"
   472  				detectedChanges = append(detectedChanges, dr)
   473  				changeCounts.Created++
   474  			}
   475  			md.ChangeState["sproc_hashes"][objIdent] = dr["md5"].(string)
   476  		}
   477  	}
   478  	// detect deletes
   479  	if !firstRun && len(md.ChangeState["sproc_hashes"]) != len(data) {
   480  		currentOidMap := make(map[string]bool)
   481  		for _, dr := range data {
   482  			currentOidMap[dr["tag_sproc"].(string)+dbMetricJoinStr+dr["tag_oid"].(string)] = true
   483  		}
   484  		for sprocIdent := range md.ChangeState["sproc_hashes"] {
   485  			_, ok := currentOidMap[sprocIdent]
   486  			if !ok {
   487  				splits := strings.Split(sprocIdent, dbMetricJoinStr)
   488  				l.WithField("sproc", splits[0]).WithField("oid", splits[1]).Debug("deleted sproc detected")
   489  				m := metrics.NewMeasurement(data.GetEpoch())
   490  				m["event"] = "drop"
   491  				m["tag_sproc"] = splits[0]
   492  				m["tag_oid"] = splits[1]
   493  				detectedChanges = append(detectedChanges, m)
   494  				delete(md.ChangeState["sproc_hashes"], sprocIdent)
   495  				changeCounts.Dropped++
   496  			}
   497  		}
   498  	}
   499  	l.Debugf("sproc changes detected: %d", len(detectedChanges))
   500  	if len(detectedChanges) > 0 {
   501  		r.measurementCh <- metrics.MeasurementEnvelope{
   502  			DBName:     md.Name,
   503  			MetricName: "sproc_changes",
   504  			Data:       detectedChanges,
   505  			CustomTags: md.CustomTags,
   506  		}
   507  	}
   508  	return changeCounts
   509  }

   511  func (r *reaper) DetectTableChanges(ctx context.Context, md *sources.DbConn) ChangeDetectionResults {
   512  	detectedChanges := make(metrics.Measurements, 0)
   513  	var firstRun bool
   514  	var changeCounts ChangeDetectionResults
   515  	l := log.GetLogger(ctx)
   516  	changeCounts.Target = "tables"

   546  				ll.Debug("new table detected")

   548  				detectedChanges = append(detectedChanges, dr)
   549  				changeCounts.Created++
   550  			}
   551  			md.ChangeState["table_hashes"][objIdent] = dr["md5"].(string)
   552  		}
   553  	}
   554  	// detect deletes
   555  	if !firstRun && len(md.ChangeState["table_hashes"]) != len(data) {
   556  		deletedTables := make([]string, 0)
   557  		// turn resultset to map => [table]=true for faster checks
   558  		currentTableMap := make(map[string]bool)

   560  			currentTableMap[dr["tag_table"].(string)] = true
   561  		}

   569  				detectedChanges = append(detectedChanges, influxEntry)
   570  				deletedTables = append(deletedTables, table)
   571  				changeCounts.Dropped++
   572  			}
   573  		}
   574  		for _, deletedTable := range deletedTables {
   575  			delete(md.ChangeState["table_hashes"], deletedTable)
   576  		}
   577  	}
   578  	l.Debugf("table changes detected: %d", len(detectedChanges))
   579  	if len(detectedChanges) > 0 {
   580  		r.measurementCh <- metrics.MeasurementEnvelope{
   581  			DBName:     md.Name,
   582  			MetricName: "table_changes",
   583  			Data:       detectedChanges,
   584  			CustomTags: md.CustomTags,
   585  		}

   669  func (r *reaper) DetectPrivilegeChanges(ctx context.Context, md *sources.DbConn) ChangeDetectionResults {
   670  	detectedChanges := make(metrics.Measurements, 0)
   671  	var firstRun bool
   672  	var changeCounts ChangeDetectionResults
   673  	l := log.GetLogger(ctx)
   674  	changeCounts.Target = "privileges"
   675  	l.Debug("checking object privilege changes...")
   676  	if _, ok := md.ChangeState["object_privileges"]; !ok {
   677  		firstRun = true
   678  		md.ChangeState["object_privileges"] = make(map[string]string)
   679  	}
   680  	mvp, ok := metricDefs.GetMetricDef("privilege_changes")
   681  	if !ok || mvp.GetSQL(int(md.Version)) == "" {
   682  		l.Warning("could not get SQL for 'privilege_changes'. cannot detect privilege changes")
   683  		return changeCounts
   684  	}
   685  	// returns rows of: object_type, tag_role, tag_object, privilege_type
   686  	data, err := QueryMeasurements(ctx, md, mvp.GetSQL(int(md.Version)))
   687  	if err != nil {
   688  		l.Error(err)
   689  		return changeCounts
   690  	}
   691  	currentState := make(map[string]bool)
   692  	for _, dr := range data {
   693  		objIdent := fmt.Sprintf("%s#:#%s#:#%s#:#%s", dr["object_type"], dr["tag_role"], dr["tag_object"], dr["privilege_type"])
   694  		ll := l.WithField("role", dr["tag_role"]).
   695  			WithField("object_type", dr["object_type"]).
   696  			WithField("object", dr["tag_object"]).
   697  			WithField("privilege_type", dr["privilege_type"])
   698  		if firstRun {
   699  			md.ChangeState["object_privileges"][objIdent] = ""
   700  		} else {
   701  			_, ok := md.ChangeState["object_privileges"][objIdent]
   702  			if !ok {
   703  				ll.Debug("new object privileges detected")
   704  				dr["event"] = "GRANT"
   705  				detectedChanges = append(detectedChanges, dr)
   706  				changeCounts.Created++
   707  				md.ChangeState["object_privileges"][objIdent] = ""
   708  			}
   709  			currentState[objIdent] = true
   710  		}
   711  	}
   712  	// check revokes - exists in old state only
   713  	if !firstRun && len(currentState) > 0 {
   714  		for objPrevRun := range md.ChangeState["object_privileges"] {
   715  			if _, ok := currentState[objPrevRun]; !ok {
   716  				splits := strings.Split(objPrevRun, "#:#")
   717  				l.WithField("role", splits[1]).
   718  					WithField("object_type", splits[0]).
   719  					WithField("object", splits[2]).
   720  					WithField("privilege_type", splits[3]).
   721  					Debug("removed object privileges detected")
   722  				revokeEntry := metrics.NewMeasurement(data.GetEpoch())
   723  				revokeEntry["object_type"] = splits[0]
   724  				revokeEntry["tag_role"] = splits[1]
   725  				revokeEntry["tag_object"] = splits[2]
   726  				revokeEntry["privilege_type"] = splits[3]
   727  				revokeEntry["event"] = "REVOKE"
   728  				detectedChanges = append(detectedChanges, revokeEntry)
   729  				changeCounts.Dropped++
   730  				delete(md.ChangeState["object_privileges"], objPrevRun)
   731  			}
   732  		}
   733  	}
   734  	l.Debugf("object privilege changes detected: %d", len(detectedChanges))
   735  	if len(detectedChanges) > 0 {
   736  		r.measurementCh <- metrics.MeasurementEnvelope{
   737  			DBName:     md.Name,
   738  			MetricName: "privilege_changes",
   739  			Data:       detectedChanges,
   740  			CustomTags: md.CustomTags,
   741  		}
   742  	}
   743  	return changeCounts
   744  }

   746  func (r *reaper) DetectConfigurationChanges(ctx context.Context, md *sources.DbConn) ChangeDetectionResults {
   747  	detectedChanges := make(metrics.Measurements, 0)
   748  	var firstRun bool
   749  	var changeCounts ChangeDetectionResults
   750  	l := log.GetLogger(ctx)
   751  	changeCounts.Target = "settings"
   752  	l.Debug("checking for configuration changes...")
   753  	if _, ok := md.ChangeState["configuration_hashes"]; !ok {
   754  		firstRun = true
   755  		md.ChangeState["configuration_hashes"] = make(map[string]string)
   756  	}
   757  	mvp, ok := metricDefs.GetMetricDef("configuration_hashes")
   758  	if !ok {
   759  		l.Error("could not get configuration_hashes sql")
   760  		return changeCounts
   761  	}
   762  	rows, err := md.Conn.Query(ctx, mvp.GetSQL(md.Version))
   763  	if err != nil {
   764  		l.Error(err)
   765  		return changeCounts
   766  	}
   767  	defer rows.Close()
   768  	var (
   769  		objIdent, objValue string
   770  		epoch              int64
   771  	)
   772  	for rows.Next() {
   773  		if rows.Scan(&epoch, &objIdent, &objValue) != nil {
   774  			return changeCounts
   775  		}
   776  		prevРash, ok := md.ChangeState["configuration_hashes"][objIdent]
   777  		ll := l.WithField("setting", objIdent)
   778  		if ok { // we have existing state
   779  			if prevРash != objValue {
   780  				ll.Warningf("settings change detected: %s = %s (prev: %s)", objIdent, objValue, prevРash)
   781  				detectedChanges = append(detectedChanges, metrics.Measurement{
   782  					metrics.EpochColumnName: epoch,
   783  					"tag_setting":           objIdent,
   784  					"value":                 objValue,
   785  					"event":                 "alter"})
   786  				md.ChangeState["configuration_hashes"][objIdent] = objValue
   787  				changeCounts.Altered++
   788  			}
   789  		} else { // check for new, delete not relevant here (pg_upgrade)
   790  			md.ChangeState["configuration_hashes"][objIdent] = objValue
   791  			if firstRun {
   792  				continue
   793  			}
   794  			ll.Debug("new setting detected")
   795  			detectedChanges = append(detectedChanges, metrics.Measurement{
   796  				metrics.EpochColumnName: epoch,
   797  				"tag_setting":           objIdent,
   798  				"value":                 objValue,
   799  				"event":                 "create"})
   800  			changeCounts.Created++
   801  		}
   802  	}
   803  	l.Debugf("configuration changes detected: %d", len(detectedChanges))
   804  	if len(detectedChanges) > 0 {
   805  		r.measurementCh <- metrics.MeasurementEnvelope{
   806  			DBName:     md.Name,
   807  			MetricName: "configuration_changes",
   808  			Data:       detectedChanges,
   809  			CustomTags: md.CustomTags,
   810  		}
   811  	}
   812  	return changeCounts
   813  }
