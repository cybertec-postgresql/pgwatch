    29  // ResolveDatabases() updates list of monitored objects from continuous monitoring sources, e.g. patroni.
    30  // Each source is resolved concurrently so that a slow or unreachable source does not block the others.
    31  func (srcs Sources) ResolveDatabases(onError func(string)) (_ SourceConns, err error) {
    32  	type result struct {
    33  		dbs SourceConns
    34  		err error
    35  	}
    36  	results := make([]result, len(srcs))
    37  	var wg sync.WaitGroup
    38  	for i, s := range srcs {
    39  		wg.Go(func() {
    40  			dbs, e := s.ResolveDatabases()
    41  			results[i] = result{dbs, e}
    42  		})
    43  	}
    44  	wg.Wait()
    45  	resolvedDbs := make(SourceConns, 0, len(srcs))
    46  	for i, res := range results {
    47  		if res.err != nil {
    48  			if onError != nil {
    49  				onError(srcs[i].Name)
    50  			}
    51  			logger.WithField("source", srcs[i].Name).WithError(res.err).Error("could not resolve databases from source")
    52  			err = errors.Join(err, res.err)
    53  		}
    54  		resolvedDbs = append(resolvedDbs, res.dbs...)
    55  	}
    56  	return resolvedDbs, err
    57  }

    97  	retmap := make(map[string]string)
    98  	if jsonText == "" {
    99  		return retmap, nil

   111  func getTransport(conf HostConfig) (*tls.Config, error) {
   112  	var caCertPool *x509.CertPool
   113  
   114  	// create valid CertPool only if the ca certificate file exists
   115  	if conf.CAFile != "" {
   116  		caCert, err := os.ReadFile(conf.CAFile)
   117  		if err != nil {
   118  			return nil, fmt.Errorf("cannot load CA file: %s", err)
   119  		}
   120  
   121  		caCertPool = x509.NewCertPool()
   122  		caCertPool.AppendCertsFromPEM(caCert)
   123  	}
   124  
   125  	var certificates []tls.Certificate
   126  
   127  	// create valid []Certificate only if the client cert and key files exists
   128  	if conf.CertFile != "" && conf.KeyFile != "" {
   129  		cert, err := tls.LoadX509KeyPair(conf.CertFile, conf.KeyFile)
   130  		if err != nil {
   131  			return nil, fmt.Errorf("cannot load client cert or key file: %s", err)
   132  		}
   133  
   134  		certificates = []tls.Certificate{cert}
   135  	}
   136  
   137  	tlsClientConfig := new(tls.Config)
   138  
   139  	if caCertPool != nil {
   140  		tlsClientConfig.RootCAs = caCertPool
   141  		if certificates != nil {
   142  			tlsClientConfig.Certificates = certificates
   143  		}
   144  	}
   145  
   146  	return tlsClientConfig, nil
   147  }

   149  func getEtcdClusterMembers(s Source, hc HostConfig) ([]PatroniClusterMember, error) {
   150  	var ret = make([]PatroniClusterMember, 0)
   151  	var cfg client.Config
   152  
   153  	if len(hc.DcsEndpoints) == 0 {
   154  		return ret, errors.New("missing ETCD connect info, make sure host config has a 'dcs_endpoints' key")
   155  	}
   156  
   157  	tlsConfig, err := getTransport(hc)
   158  	if err != nil {
   159  		return nil, err
   160  	}
   161  	cfg = client.Config{
   162  		Endpoints:            hc.DcsEndpoints,
   163  		TLS:                  tlsConfig,
   164  		DialKeepAliveTimeout: time.Second,
   165  		Username:             hc.Username,
   166  		Password:             hc.Password,
   167  		DialTimeout:          5 * time.Second,
   168  		Logger:               zap.NewNop(),
   169  	}
   170  
   171  	c, err := client.New(cfg)
   172  	if err != nil {
   173  		return ret, err
   174  	}
   175  	defer c.Close()
   176  
   177  	ctx, cancel := context.WithTimeoutCause(context.Background(), 5*time.Second, errors.New("etcd client timeout"))
   178  	defer cancel()
   179  
   180  	// etcd3 does not have a dir node.
   181  	// Key="/namespace/scope/leader", e.g. "/service/batman/leader"
   182  	// Key="/namespace/scope/members/node", e.g. "/service/batman/members/pg1"
   183  
   184  	resp, err := c.Get(ctx, hc.Path, client.WithPrefix())
   185  	if err != nil {
   186  		return ret, cmp.Or(context.Cause(ctx), err)
   187  	}
   188  
   189  	for _, node := range resp.Kvs {
   190  		// remove leading slash and split by "/"
   191  		parts := strings.Split(strings.TrimPrefix(string(node.Key), "/"), "/")
   192  		if len(parts) < 4 || parts[2] != "members" {
   193  			continue // skip non-member keys
   194  		}
   195  		nodeData, err := jsonTextToStringMap(string(node.Value))
   196  		if err != nil {
   197  			logger.Errorf("Could not parse ETCD node data for node \"%s\": %s", node.Key, err)
   198  			continue
   199  		}
   200  		role := nodeData["role"]
   201  		connURL := nodeData["conn_url"]
   202  		scope := parts[1]
   203  		name := parts[3]
   204  		ret = append(ret, PatroniClusterMember{Scope: scope, ConnURL: connURL, Role: role, Name: name})
   205  	}
   206  
   207  	lastFoundClusterMembers[s.Name] = ret
   208  	return ret, nil
   209  }

   234  func NewHostConfig(URI string) (hc HostConfig, err error) {
   235  	// Extract scheme
   236  	before, after, ok := strings.Cut(URI, "://")
   237  	if !ok {
   238  		return hc, fmt.Errorf("invalid URI: missing scheme")
   239  	}
   240  	scheme := before
   241  	remainder := after // skip "://"
   242  
   243  	// Find where the host portion ends (at first '/' or '?' or end of string)
   244  	hostEnd := strings.IndexAny(remainder, "/?")
   245  	var hostPart, pathAndQuery string
   246  	if hostEnd == -1 {
   247  		hostPart = remainder
   248  		pathAndQuery = ""
   249  	} else {
   250  		hostPart = remainder[:hostEnd]
   251  		pathAndQuery = remainder[hostEnd:]
   252  	}
   253  
   254  	// Check for user info (username:password@)
   255  	var userInfo string
   256  	if atIdx := strings.LastIndex(hostPart, "@"); atIdx != -1 {
   257  		userInfo = hostPart[:atIdx]
   258  		hostPart = hostPart[atIdx+1:]
   259  	}
   260  
   261  	// Split hosts by comma for multiple endpoints
   262  	hosts := strings.Split(hostPart, ",")
   263  
   264  	// Parse a clean URL with just the first host to extract other components
   265  	cleanURI := scheme + "://"
   266  	if userInfo != "" {
   267  		cleanURI += userInfo + "@"
   268  	}
   269  	cleanURI += hosts[0] + pathAndQuery
   270  
   271  	var url *url.URL
   272  	url, err = url.Parse(cleanURI)
   273  	if err != nil {
   274  		return
   275  	}
   276  
   277  	switch url.Scheme {
   278  	case dcsTypeEtcd:
   279  		hc.DcsType = dcsTypeEtcd
   280  		for _, h := range hosts {
   281  			hc.DcsEndpoints = append(hc.DcsEndpoints, "http://"+h)
   282  		}
   283  	case dcsTypeZookeeper:
   284  		hc.DcsType = dcsTypeZookeeper
   285  		hc.DcsEndpoints = hosts // Use the split hosts directly
   286  	case dcsTypeConsul:
   287  		hc.DcsType = dcsTypeConsul
   288  		hc.DcsEndpoints = hosts // Use the split hosts directly
   289  	default:
   290  		return hc, fmt.Errorf("unsupported DCS type: %s", url.Scheme)
   291  	}
   292  
   293  	hc.Path = url.Path
   294  	hc.Username = url.User.Username()
   295  	hc.Password, _ = url.User.Password() // password is optional, so we ignore the error
   296  	hc.CAFile = url.Query().Get("ca_file")
   297  	hc.CertFile = url.Query().Get("cert_file")
   298  	hc.KeyFile = url.Query().Get("key_file")
   299  
   300  	return hc, nil
   301  }

   303  func ResolveDatabasesFromPatroni(source Source) (SourceConns, error) {
   304  	var mds SourceConns
   305  	var clusterMembers []PatroniClusterMember
   306  	var err error
   307  	var ok bool
   308  
   309  	hostConfig, err := NewHostConfig(source.ConnStr)
   310  	if err != nil {
   311  		return nil, err
   312  	}
   313  
   314  	switch hostConfig.DcsType {
   315  	case dcsTypeEtcd:
   316  		clusterMembers, err = getEtcdClusterMembers(source, hostConfig)
   317  	case dcsTypeZookeeper:
   318  		clusterMembers, err = getZookeeperClusterMembers(source, hostConfig)
   319  	case dcsTypeConsul:
   320  		clusterMembers, err = getConsulClusterMembers(source)
   321  	default:
   322  		return nil, errors.New("unknown DCS")
   323  	}
   324  	logger := logger.WithField("source", source.Name)
   325  	if err != nil {
   326  		if errors.Is(err, errors.ErrUnsupported) {
   327  			return nil, err
   328  		}
   329  		logger.Debug("failed to get info from DCS, using previous member info if any")
   330  		if clusterMembers, ok = lastFoundClusterMembers[source.Name]; ok { // mask error from main loop not to remove monitored DBs due to "jitter"
   331  			err = nil
   332  		}
   333  	} else {
   334  		lastFoundClusterMembers[source.Name] = clusterMembers
   335  	}
   336  	if len(clusterMembers) == 0 {
   337  		return mds, err
   338  	}
   339  
   340  	for _, patroniMember := range clusterMembers {
   341  		logger.Info("processing Patroni cluster member: ", patroniMember.Name)
   342  		if source.OnlyIfMaster && !patroniMember.IsPrimary() {
   343  			continue
   344  		}
   345  		src := *source.Clone()
   346  		src.ConnStr = patroniMember.ConnURL
   347  		if !hostConfig.IsScopeSpecified() {
   348  			src.Name += "_" + patroniMember.Scope
   349  		}
   350  		src.Name += "_" + patroniMember.Name
   351  		if dbs, err := ResolveDatabasesFromPostgres(src); err == nil {
   352  			mds = append(mds, dbs...)
   353  		} else {
   354  			logger.WithError(err).Error("failed to resolve databases for Patroni member: ", patroniMember.Name)
   355  		}
   356  	}
   357  	return mds, err
   358  }

   360  // ResolveDatabasesFromPostgres reads all the databases from the given cluster,
   361  // additionally matching/not matching specified regex patterns
   362  func ResolveDatabasesFromPostgres(s Source) (resolvedDbs SourceConns, err error) {
   363  	var (
   364  		c      db.PgxPoolIface
   365  		dbname string
   366  		rows   pgx.Rows
   367  	)
   368  	c, err = NewConn(context.TODO(), s.ConnStr)
   369  	if err != nil {
   370  		return
   371  	}
   372  	defer c.Close()
   373  
   374  	sql := `select /* pgwatch_generated */
   375  	datname
   376  	from pg_database
   377  	where not datistemplate
   378  	and datallowconn
   379  	and has_database_privilege (datname, 'CONNECT')
   380  	and case when length(trim($1)) > 0 then datname ~ $1 else true end
   381  	and case when length(trim($2)) > 0 then not datname ~ $2 else true end`
   382  
   383  	if rows, err = c.Query(context.TODO(), sql, s.IncludePattern, s.ExcludePattern); err != nil {
   384  		return nil, err
   385  	}
   386  	for rows.Next() {
   387  		if err = rows.Scan(&dbname); err != nil {
   388  			return nil, err
   389  		}
   390  		rdb := NewDbConn(*s.Clone())
   391  		rdb.Name += "_" + dbname
   392  		rdb.SetDatabaseName(dbname)
   393  		resolvedDbs = append(resolvedDbs, rdb)
   394  	}
   395  
   396  	if err := rows.Err(); err != nil {
   397  		return nil, err
   398  	}
   399  	return
   400  }
