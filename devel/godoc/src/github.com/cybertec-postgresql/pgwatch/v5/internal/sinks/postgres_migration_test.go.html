    15  // TestMigrationsCountInvariant is a fast, dependency-free unit test that guards against the
    16  // "stale MigrationsCount" class of bug. It asserts that MigrationsCount matches:
    17  //   - the number of migrations actually registered in migrations(), and
    18  //   - the number of rows seeded into admin.migration in admin_schema.sql.
    19  //
    20  // Whenever a migration is added, all three must be bumped together, so this test fails loudly
    21  // if any of them drifts out of sync.
    22  func TestMigrationsCountInvariant(t *testing.T) {
    23  	a := assert.New(t)
    24  
    25  	// 1. Count migrations actually registered in migrations().
    26  	m, err := migrator.New(migrations())
    27  	require.NoError(t, err)
    28  	registered := m.Count()
    29  	a.Equal(MigrationsCount, registered,
    30  		"MigrationsCount (%d) must equal the number of migrations registered in migrations() (%d)",
    31  		MigrationsCount, registered)
    32  
    33  	// 2. Count rows seeded into admin.migration in admin_schema.sql. The seed looks like:
    34  	//     INSERT INTO admin.migration (id, version) VALUES
    35  	//         (0, '...'),
    36  	//         (1, '...'),
    37  	//         (2, '...');
    38  	seeded := len(regexp.MustCompile(`(?m)^\s*\(\d+,\s*'`).FindAllString(sqlMetricAdminSchema, -1))
    39  	a.Equal(MigrationsCount, seeded,
    40  		"MigrationsCount (%d) must equal the number of rows seeded into admin.migration in admin_schema.sql (%d)",
    41  		MigrationsCount, seeded)
    42  }

    44  // simulatePreV6Migration rolls the seeded admin.migration table back to a pre-01409 state so
    45  // that the migrator considers the "01409 Switch to time-only partitioning" migration pending.
    46  // A fresh NewPostgresSinkMigrator bootstrap seeds all migrations as applied (the current install
    47  // path); here we emulate an older database that must still be upgraded.
    48  //
    49  // The migrator determines pending migrations purely from the row count in admin.migration, so we
    50  // delete the last-applied migration row, leaving MigrationsCount-1 rows behind.
    51  func simulatePreV6Migration(t *testing.T, conn *pgx.Conn) {
    52  	t.Helper()
    53  	_, err := conn.Exec(ctx, `DELETE FROM admin.migration WHERE id = (SELECT max(id) FROM admin.migration)`)
    54  	require.NoError(t, err)
    55  
    56  	var count int
    57  	require.NoError(t, conn.QueryRow(ctx, `SELECT count(*) FROM admin.migration`).Scan(&count))
    58  	require.Equal(t, MigrationsCount-1, count, "exactly one migration should be pending after rollback")
    59  }

    61  // oldSchemaMetricTable creates a metric table using the pre-v6 (dbname -> time) two-level
    62  // partitioning layout that the "01409 Switch to time-only partitioning" migration converts from:
    63  //
    64  //	public.<metric>                      PARTITION BY LIST (dbname)      -- top level
    65  //	  subpartitions.<metric>_<dbname>    PARTITION BY RANGE (time)       -- dbname level
    66  //	    subpartitions.<metric>_<...>_<w> leaf partition                  -- time level
    67  //
    68  // It seeds a couple of rows so the test can assert data survives the migration.
    69  func oldSchemaMetricTable(t *testing.T, conn *pgx.Conn, metric string) {
    70  	t.Helper()
    71  	_, err := conn.Exec(ctx, `
    72  		CREATE TABLE public.`+metric+` (LIKE admin.metrics_template INCLUDING INDEXES) PARTITION BY LIST (dbname);
    73  		COMMENT ON TABLE public.`+metric+` IS 'pgwatch-generated-metric-lvl';
    74  
    75  		CREATE TABLE subpartitions.`+metric+`_db1 PARTITION OF public.`+metric+`
    76  			FOR VALUES IN ('db1') PARTITION BY RANGE (time);
    77  
    78  		CREATE TABLE subpartitions.`+metric+`_db1_2024w01 PARTITION OF subpartitions.`+metric+`_db1
    79  			FOR VALUES FROM ('2024-01-01') TO ('2024-01-08');
    80  		COMMENT ON TABLE subpartitions.`+metric+`_db1_2024w01 IS 'pgwatch-generated-metric-time-lvl';
    81  
    82  		INSERT INTO public.`+metric+` (time, dbname, data) VALUES
    83  			('2024-01-03 10:00:00+00', 'db1', '{"x": 1}'::jsonb),
    84  			('2024-01-04 11:00:00+00', 'db1', '{"x": 2}'::jsonb);
    85  	`)
    86  	require.NoError(t, err)
    87  }

    89  // isRangePartitioned reports whether the given relation is now top-level RANGE partitioned.
    90  func isRangePartitioned(t *testing.T, conn *pgx.Conn, metric string) bool {
    91  	t.Helper()

   108  // TestMigration01409_TimeOnlyPartitioning is an end-to-end test against a real PostgreSQL
   109  // container. It builds the old (dbname -> time) partitioned layout, runs the sink migrations,
   110  // and asserts that the table is converted to time-only RANGE partitioning while preserving data.
   111  // It also runs the migration a second time to verify idempotency / re-run safety.
   112  func TestMigration01409_TimeOnlyPartitioning(t *testing.T) {
   113  	if os.Getenv("PGWATCH_TEST_SKIP_MIGRATION") != "" {
   114  		t.Skip("migration integration test skipped via PGWATCH_TEST_SKIP_MIGRATION")
   115  	}
   116  	r := require.New(t)
   117  	a := assert.New(t)
   118  
   119  	pgContainer, pgTearDown, err := testutil.SetupPostgresContainer()
   120  	r.NoError(err)
   121  	defer pgTearDown()
   122  
   123  	connStr, err := pgContainer.ConnectionString(ctx, "sslmode=disable")
   124  	r.NoError(err)
   125  
   126  	// Bootstrap the admin schema (creates admin.*, subpartitions, metrics_template, etc.).
   127  	// A fresh bootstrap seeds admin.migration with ALL migrations already applied, so we
   128  	// delete the 01409 row to simulate an older database that predates this migration.
   129  	mig, err := NewPostgresSinkMigrator(ctx, connStr)
   130  	r.NoError(err)
   131  
   132  	conn, err := pgx.Connect(ctx, connStr)
   133  	r.NoError(err)
   134  	defer conn.Close(ctx)
   135  
   136  	simulatePreV6Migration(t, conn)
   137  
   138  	const metric = "old_style_metric"
   139  	oldSchemaMetricTable(t, conn, metric)
   140  
   141  	// Sanity check: before migration the table is LIST (dbname) partitioned, not RANGE.
   142  	a.False(isRangePartitioned(t, conn, metric), "table should start as LIST(dbname) partitioned")
   143  	a.Equal(2, rowCount(t, conn, metric), "seeded rows should be present before migration")
   144  
   145  	// Run the migrations (this executes 01110, 01180 and the 01409 conversion).
   146  	r.NoError(mig.Migrate())
   147  
   148  	// After migration the top-level table must be RANGE(time) partitioned and keep its data.
   149  	a.True(isRangePartitioned(t, conn, metric), "table should be converted to RANGE(time) partitioning")
   150  	a.Equal(2, rowCount(t, conn, metric), "data must be preserved across the migration")
   151  
   152  	// The temporary *_before_v6_migration table must have been cleaned up.
   153  	var leftover bool
   154  	r.NoError(conn.QueryRow(ctx,
   155  		`SELECT to_regclass($1) IS NOT NULL`, metric+"_before_v6_migration").Scan(&leftover))
   156  	a.False(leftover, "the *_before_v6_migration scratch table should be dropped")
   157  
   158  	// Idempotency: running the migrations again must not error and must not lose data.
   159  	needs, err := mig.NeedsMigration()
   160  	r.NoError(err)
   161  	a.False(needs, "no migrations should be pending immediately after a successful migrate")
   162  
   163  	r.NoError(mig.Migrate(), "re-running Migrate() must be a no-op and not error")
   164  	a.True(isRangePartitioned(t, conn, metric))
   165  	a.Equal(2, rowCount(t, conn, metric), "data must remain intact after a second migrate")
   166  }
