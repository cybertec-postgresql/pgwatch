   236  // VersionToInt parses a given version and returns an integer  or
   237  // an error if unable to parse the version. Only parses valid semantic versions.
   238  // Performs checking that can find errors within the version.
   239  // Examples: v1.2 -> 01_02_00, v9.6.3 -> 09_06_03, v11 -> 11_00_00
   240  var regVer = regexp.MustCompile(`(\d+).?(\d*).?(\d*)`)
   241  
   242  func VersionToInt(version string) (v int) {
   243  	if matches := regVer.FindStringSubmatch(version); len(matches) > 1 {
   244  		for i, match := range matches[1:] {
   245  			v += func() (m int) { m, _ = strconv.Atoi(match); return }() * int(math.Pow10(4-i*2))
   246  		}
   247  	}
   248  	return
   249  }

   251  func (md *DbConn) FetchRuntimeInfo(ctx context.Context, forceRefetch bool) (err error) {
   252  	// Fast path: check the atomic timestamp without acquiring any lock.
   253  	// This avoids lock contention when the cached value is still fresh.
   254  	if !forceRefetch && time.Duration(time.Now().UnixNano()-md.lastCheckedNs.Load()) < 5*time.Minute {
   255  		return nil
   256  	}
   257  
   258  	md.Lock()
   259  	defer md.Unlock()
   260  	if ctx.Err() != nil {
   261  		return ctx.Err()
   262  	}
   263  
   264  	switch md.Kind {
   265  	case SourcePgBouncer, SourcePgPool:
   266  		if md.VersionStr, md.Version, err = md.FetchVersion(ctx, func() string {
   267  			if md.Kind == SourcePgBouncer {
   268  				return "SHOW VERSION"
   269  			}
   270  			return "SHOW POOL_VERSION"
   271  		}()); err != nil {
   272  			return
   273  		}
   274  	default:
   275  		sql := `select /* pgwatch_generated */ 
   276  	div(current_setting('server_version_num')::int, 10000) as ver, 
   277  	version(), 
   278  	pg_is_in_recovery(), 
   279  	current_database()::TEXT,
   280  	system_identifier,
   281  	current_setting('is_superuser')::bool
   282  FROM
   283  	pg_control_system()`
   284  
   285  		err = md.Conn.QueryRow(ctx, sql).
   286  			Scan(&md.Version, &md.VersionStr,
   287  				&md.IsInRecovery, &md.RealDbname,
   288  				&md.SystemIdentifier, &md.IsSuperuser)
   289  		if err != nil {
   290  			return err
   291  		}
   292  
   293  		md.ExecEnv = md.DiscoverPlatform(ctx)
   294  		md.ApproxDbSize = md.FetchApproxSize(ctx)
   295  
   296  		sqlExtensions := `select /* pgwatch_generated */ extname::text, (regexp_matches(extversion, $$\d+\.?\d+?$$))[1]::text as extversion from pg_extension order by 1;`
   297  		var res pgx.Rows
   298  		res, err = md.Conn.Query(ctx, sqlExtensions)
   299  		if err == nil {
   300  			var ext string
   301  			var ver string
   302  			_, err = pgx.ForEachRow(res, []any{&ext, &ver}, func() error {
   303  				extver := VersionToInt(ver)
   304  				if extver == 0 {
   305  					return fmt.Errorf("unexpected extension %s version input: %s", ext, ver)
   306  				}
   307  				md.Extensions[ext] = extver
   308  				return nil
   309  			})
   310  		}
   311  
   312  	}
   313  	md.lastCheckedNs.Store(time.Now().UnixNano())
   314  	return err
   315  }

   361  // TryCreateMissingExtensions should be called once on daemon startup if some commonly wanted extension (most notably pg_stat_statements) is missing.
   362  func (md *DbConn) TryCreateMissingExtensions(ctx context.Context, extensions []string) (string, error) {
   363  	// Snapshot the already-known extensions under RLock; release before doing any I/O.
   364  	md.RLock()
   365  	knownExts := maps.Clone(md.Extensions)
   366  	md.RUnlock()
   367  
   368  	sqlAvailableExts := `select name::text from pg_available_extensions order by 1`
   369  	createdExts := make([]string, 0)
   370  
   371  	data, err := md.Conn.Query(ctx, sqlAvailableExts)
   372  	if err != nil {
   373  		return "", err
   374  	}
   375  	availableExts, err := pgx.CollectRows(data, pgx.RowTo[string])
   376  	if err != nil {
   377  		return "", err
   378  	}
   379  
   380  	for _, extToCreate := range extensions {
   381  		if _, ok := knownExts[extToCreate]; ok {
   382  			continue
   383  		}
   384  		if _, ok := slices.BinarySearch(availableExts, extToCreate); !ok {
   385  			err = errors.Join(err, fmt.Errorf("requested extension %s is not available on instance", extToCreate))
   386  			continue
   387  		}
   388  		if _, e := md.Conn.Exec(ctx, fmt.Sprintf(`create extension if not exists "%s"`, extToCreate)); e != nil {
   389  			err = errors.Join(err, fmt.Errorf("failed to create extension %s: %w", extToCreate, e))
   390  		} else {
   391  			createdExts = append(createdExts, extToCreate)
   392  		}
   393  	}
   394  	return strings.Join(createdExts, ","), err
   395  }

   465  // ParseConfig parses pc.ConnStr once and caches the result in pc.connConfig.
   466  // Subsequent calls are no-ops. Mirrors DbConn.ParseConfig.
   467  func (pc *PromConn) ParseConfig() error {
   468  	if pc.connConfig != nil {
   469  		return nil
   470  	}
   471  	u, err := url.Parse(pc.ConnStr)
   472  	if err != nil {
   473  		return fmt.Errorf("parsing prometheus source URL: %w", err)
   474  	}
   475  	userinfo := u.User
   476  	u.User = nil
   477  
   478  	q := u.Query()
   479  	tlsRootCert := q.Get("tlsrootcert")
   480  	tlsSkipVerify := q.Get("tlsskipverify") == "true"
   481  	q.Del("tlsrootcert")
   482  	q.Del("tlsskipverify")
   483  	u.RawQuery = q.Encode()
   484  
   485  	tlsConfig := &tls.Config{InsecureSkipVerify: tlsSkipVerify} //nolint:gosec // intentional per config
   486  	if tlsRootCert != "" {
   487  		caCert, readErr := os.ReadFile(tlsRootCert)
   488  		if readErr != nil {
   489  			return fmt.Errorf("reading tlsrootcert %q: %w", tlsRootCert, readErr)
   490  		}
   491  		pool := x509.NewCertPool()
   492  		pool.AppendCertsFromPEM(caCert)
   493  		tlsConfig.RootCAs = pool
   494  	}
   495  
   496  	pc.connConfig = &promConnConfig{
   497  		URL:       u.String(),
   498  		Userinfo:  userinfo,
   499  		TLSConfig: tlsConfig,
   500  	}
   501  	return nil
   502  }

   504  func (pc *PromConn) Connect(ctx context.Context, _ CmdOpts) error {
   505  	pc.Lock()
   506  	if pc.HTTPClient == nil {
   507  		if err := pc.ParseConfig(); err != nil {
   508  			pc.Unlock()
   509  			return err
   510  		}
   511  		pc.HTTPClient = &http.Client{
   512  			Transport: &http.Transport{TLSClientConfig: pc.connConfig.TLSConfig},
   513  			Timeout:   30 * time.Second,
   514  			CheckRedirect: func(*http.Request, []*http.Request) error {
   515  				return http.ErrUseLastResponse
   516  			},
   517  		}
   518  	}
   519  	pc.Unlock()
   520  	return pc.Ping(ctx)
   521  }

   523  // Scrape executes a single GET request to the source's metrics endpoint with
   524  // Accept: text/plain and optional Basic Auth from the cached config.
   525  // The caller is responsible for closing resp.Body.
   526  // Connect must be called before Scrape.
   527  func (pc *PromConn) Scrape(ctx context.Context) (*http.Response, error) {
   528  	pc.RLock()
   529  	client := pc.HTTPClient
   530  	cfg := pc.connConfig
   531  	pc.RUnlock()
   532  
   533  	if client == nil || cfg == nil {
   534  		return nil, errors.New("prometheus source not connected: call Connect first")
   535  	}
   536  
   537  	req, err := http.NewRequestWithContext(ctx, http.MethodGet, cfg.URL, nil)
   538  	if err != nil {
   539  		return nil, err
   540  	}
   541  	req.Header.Set("Accept", "text/plain")
   542  	if cfg.Userinfo != nil {
   543  		pass, _ := cfg.Userinfo.Password()
   544  		req.SetBasicAuth(cfg.Userinfo.Username(), pass)
   545  	}
   546  	return client.Do(req)
   547  }
