    86  // Reap() starts the main monitoring loop. It is responsible for fetching metrics measurements
    87  // from the sources and storing them to the sinks. It also manages the lifecycle of
    88  // the metric gatherers. In case of a source or metric definition change, it will
    89  // start or stop the gatherers accordingly.
    90  func (r *reaper) Reap(ctx context.Context) {
    91  	var err error
    92  
    93  	go r.WriteMeasurements(ctx)
    94  
    95  	r.ready.Store(true)
    96  
    97  	for { //main loop
    98  		if r.Logging.LogLevel == "debug" {
    99  			r.PrintMemStats()
   100  		}
   101  		if err = r.LoadSources(ctx); err != nil {
   102  			r.logger.Error("could not refresh active sources, using last valid cache: %w", err)
   103  		}
   104  		if err = r.LoadMetrics(); err != nil {
   105  			r.logger.Error("could not refresh metric definitions, using last valid cache: %w", err)
   106  		}
   107  
   108  		for _, monitoredSource := range r.monitoredSources {
   109  			src := monitoredSource.GetSource()
   110  			srcL := r.logger.WithField("source", src.Name)
   111  			ctx = log.WithLogger(ctx, srcL)
   112  
   113  			if monitoredSource.Connect(ctx, r.Sources) != nil {
   114  				r.WriteInstanceDown(src.Name)
   115  				srcL.Warning("could not init connection, retrying on next iteration")
   116  				continue
   117  			}
   118  
   119  			switch md := monitoredSource.(type) {
   120  			case *sources.DbConn:
   121  				if err = md.FetchRuntimeInfo(ctx, true); err != nil {
   122  					srcL.Error("could not start metric gathering: %w", err)
   123  					continue
   124  				}
   125  				if r.FilterSource(ctx, md) {
   126  					continue
   127  				}
   128  				r.CreateSourceHelpers(ctx, md)
   129  				r.TrackRecoveryStatus(ctx, md)
   130  				r.SyncMetricsToSinks(ctx, md)
   131  				r.StartWorker(ctx, src.Name, NewDbConnReaper(r, md))
   132  			case *sources.PromConn:
   133  				r.StartWorker(ctx, src.Name, NewPromSourceReaper(r, md))
   134  			}
   135  		}
   136  		r.CleanupRemovedWorkers(ctx)
   137  		select {
   138  		case <-time.After(time.Second * time.Duration(r.Sources.Refresh)):
   139  			r.logger.Debugf("wake up after %d seconds", r.Sources.Refresh)
   140  		case <-ctx.Done():
   141  			return
   142  		}
   143  	}
   144  }

   146  // StartWorker launches a source reaper goroutine for the given source if one
   147  // is not already running. It is a no-op when a worker for that name exists.
   148  func (r *reaper) StartWorker(ctx context.Context, sourceName string, sr Reaper) {
   149  	if _, exists := r.cancelFuncs[sourceName]; exists {
   150  		return
   151  	}
   152  	log.GetLogger(ctx).Info("starting source reaper")
   153  	sourceCtx, cancelFunc := context.WithCancel(ctx)
   154  	r.cancelFuncs[sourceName] = cancelFunc
   155  	go sr.Reap(sourceCtx)
   156  }

   158  // FilterSource snapshots the mutable RuntimeInfo under a single RLock, logs
   159  // the connection status, and applies both eligibility filters:
   160  //   - OnlyIfMaster: skip standby sources that must not be monitored as standbys.
   161  //   - MinDbSizeMB:  skip sources whose database is below the configured size threshold.
   162  //
   163  // Returns true when the source should be skipped for this loop iteration.
   164  func (r *reaper) FilterSource(ctx context.Context, md *sources.DbConn) bool {
   165  	md.RLock()
   166  	isInRecovery := md.IsInRecovery
   167  	versionStr := md.VersionStr
   168  	DBSizeMB := md.ApproxDbSize / 1048576
   169  	md.RUnlock()
   170  
   171  	l := log.GetLogger(ctx)
   172  
   173  	if isInRecovery && md.OnlyIfMaster {
   174  		l.Info("not added to monitoring due to 'master only' property and status change")
   175  		r.ShutdownWorker(ctx, md.Name)
   176  		return true
   177  	}
   178  
   179  	if DBSizeMB != 0 && DBSizeMB < r.Sources.MinDbSizeMB {
   180  		l.Infof("ignored due to the --min-db-size-mb filter, current size %d MB", DBSizeMB)
   181  		r.ShutdownWorker(ctx, md.Name)
   182  		return true
   183  	}
   184  
   185  	l.WithField("recovery", isInRecovery).Infof("Connect OK. Version: %s", versionStr)
   186  	return false
   187  }

   189  // TrackRecoveryStatus logs any primary/standby role changes and updates the
   190  // per-source recovery-status cache.
   191  func (r *reaper) TrackRecoveryStatus(ctx context.Context, md *sources.DbConn) {
   192  	md.RLock()
   193  	isInRecovery := md.IsInRecovery
   194  	hasStandbyConfig := len(md.MetricsStandby) > 0
   195  	md.RUnlock()
   196  
   197  	l := log.GetLogger(ctx)
   198  	if r.srcRecoveryStatus[md.Name] != isInRecovery {
   199  		if isInRecovery && hasStandbyConfig {
   200  			l.Warning("Switching metrics collection to standby config...")
   201  		} else if !isInRecovery {
   202  			l.Warning("Switching metrics collection to primary config...")
   203  		}
   204  		// else: standby without a dedicated standby config keeps primary config, no warn
   205  	}
   206  	r.srcRecoveryStatus[md.Name] = isInRecovery
   207  }

   209  // SyncMetricsToSinks syncs metric names with sinks for the active config
   210  func (r *reaper) SyncMetricsToSinks(ctx context.Context, md *sources.DbConn) {
   211  	l := log.GetLogger(ctx)
   212  	for metricName := range md.ActiveMetrics() {
   213  		mvp, metricDefExists := metricDefs.GetMetricDef(metricName)
   214  		if !metricDefExists {
   215  			epoch, ok := lastSQLFetchError.Load(metricName)
   216  			if !ok || ((time.Now().Unix() - epoch.(int64)) > 3600) {
   217  				l.WithField("metric", metricName).Warning("metric definition not found")
   218  				lastSQLFetchError.Store(metricName, time.Now().Unix())
   219  			}
   220  			continue
   221  		}
   222  		metricNameForStorage := metricName
   223  		if !r.isSpecialMetric(metricName) && mvp.StorageName > "" {
   224  			metricNameForStorage = mvp.StorageName
   225  		}
   226  		if err := r.SinksWriter.SyncMetric(md.Name, metricNameForStorage, sinks.AddOp); err != nil {
   227  			l.Error(err)
   228  		}
   229  	}
   230  }

   232  // CreateSourceHelpers creates the extensions and metric helpers for the monitored source
   233  func (r *reaper) CreateSourceHelpers(ctx context.Context, monitoredSource *sources.DbConn) {
   234  	if r.prevLoopMonitoredDBs.GetMonitoredDatabase(monitoredSource.Name) != nil {
   235  		return // already created
   236  	}
   237  	monitoredSource.RLock()
   238  	isInRecovery := monitoredSource.IsInRecovery
   239  	monitoredSource.RUnlock()
   240  	if !monitoredSource.IsPostgresSource() || isInRecovery {
   241  		return // no need to create anything for non-postgres sources
   242  	}
   243  
   244  	l := log.GetLogger(ctx)
   245  	if r.Sources.TryCreateListedExtsIfMissing > "" {
   246  		l.Info("trying to create extensions if missing")
   247  		extsToCreate := strings.Split(r.Sources.TryCreateListedExtsIfMissing, ",")
   248  		extsCreated, err := monitoredSource.TryCreateMissingExtensions(ctx, extsToCreate)
   249  		if err != nil {
   250  			l.Warning(err)
   251  		}
   252  		if extsCreated != "" {
   253  			l.Infof("%d/%d extensions created: %s", len(extsCreated), len(extsToCreate), extsCreated)
   254  		}
   255  	}
   256  
   257  	if r.Sources.CreateHelpers {
   258  		l.Info("trying to create helper objects if missing")
   259  		if err := monitoredSource.TryCreateMetricsHelpers(ctx, func(metric string) string {
   260  			if m, ok := metricDefs.GetMetricDef(metric); ok {
   261  				return m.InitSQL
   262  			}
   263  			return ""
   264  		}); err != nil {
   265  			l.Warning(err)
   266  		}
   267  	}
   268  }

   276  // ShutdownWorker stops the source reaper for a single named source, closes its
   277  // connection pool, and deregisters it from the sinks.
   278  func (r *reaper) ShutdownWorker(_ context.Context, sourceName string) {
   279  	if cancelFunc, exists := r.cancelFuncs[sourceName]; exists {
   280  		r.logger.WithField("source", sourceName).Info("stopping source reaper...")
   281  		cancelFunc()
   282  		delete(r.cancelFuncs, sourceName)
   283  	}
   284  	if db := r.monitoredSources.GetMonitoredDatabase(sourceName); db != nil {
   285  		db.Close()
   286  	}
   287  	if err := r.SinksWriter.SyncMetric(sourceName, "", sinks.DeleteOp); err != nil {
   288  		r.logger.Error(err)
   289  	}
   290  }

   292  // CleanupRemovedWorkers stops workers for sources that are no longer in
   293  // monitoredSources or whose context has been cancelled, and closes connections
   294  // for any sources that disappeared from the previous loop without a running worker.
   295  func (r *reaper) CleanupRemovedWorkers(ctx context.Context) {
   296  	r.logger.Debug("checking if any workers need to be shut down...")
   297  	for sourceName := range r.cancelFuncs {
   298  		md := r.monitoredSources.GetMonitoredDatabase(sourceName)
   299  		if ctx.Err() == nil && md != nil {
   300  			continue // source still active
   301  		}
   302  		if md == nil {
   303  			r.logger.Debugf("Source %s removed from config, shutting down source reaper...", sourceName)
   304  		}
   305  		r.ShutdownWorker(ctx, sourceName)
   306  	}
   307  	// Close connections for sources that disappeared without ever having a worker.
   308  	for _, prevDB := range r.prevLoopMonitoredDBs {
   309  		if r.monitoredSources.GetMonitoredDatabase(prevDB.GetSource().Name) == nil {
   310  			prevDB.Close()
   311  			_ = r.SinksWriter.SyncMetric(prevDB.GetSource().Name, "", sinks.DeleteOp)
   312  		}
   313  	}
   314  	r.prevLoopMonitoredDBs = slices.Clone(r.monitoredSources)
   315  }

   317  // LoadSources loads sources from the reader
   318  func (r *reaper) LoadSources(ctx context.Context) (err error) {
   319  	if DoesEmergencyTriggerfileExist(r.Metrics.EmergencyPauseTriggerfile) {
   320  		r.logger.Warningf("Emergency pause triggerfile detected at %s, ignoring currently configured DBs", r.Metrics.EmergencyPauseTriggerfile)
   321  		r.monitoredSources = make(sources.SourceConns, 0)
   322  		return nil
   323  	}
   324  
   325  	var newSrcs sources.SourceConns
   326  	srcs, err := r.SourcesReaderWriter.GetSources()
   327  	if err != nil {
   328  		return err
   329  	}
   330  	srcs = slices.DeleteFunc(srcs, func(s sources.Source) bool {
   331  		// filter out disabled sources and sources with group not in the list of groups to monitor
   332  		return !s.IsEnabled || len(r.Sources.Groups) > 0 && !slices.Contains(r.Sources.Groups, s.Group)
   333  	})
   334  
   335  	if newSrcs, err = srcs.ResolveDatabases(r.WriteInstanceDown); err != nil {
   336  		// discover dtabases for continuous monitoring sources
   337  		r.logger.WithError(err).Error("could not resolve databases from sources")
   338  	}
   339  
   340  	for i, newMD := range newSrcs {
   341  		md := r.monitoredSources.GetMonitoredDatabase(newMD.GetSource().Name)
   342  		if md == nil {
   343  			continue
   344  		}
   345  		if md.GetSource().Equal(newMD.GetSource()) {
   346  			// replace with the existing connection if the source is the same
   347  			newSrcs[i] = md
   348  			continue
   349  		}
   350  		// Source configs changed, stop all running gatherers to trigger a restart
   351  		// TODO: Optimize this for single metric addition/deletion/interval-change cases to not do a full restart
   352  		r.logger.WithField("source", md.GetSource().Name).Info("Source configs changed, restarting all gatherers...")
   353  		r.ShutdownWorker(ctx, md.GetSource().Name)
   354  	}
   355  	r.monitoredSources = newSrcs
   356  	r.logger.WithField("sources", len(r.monitoredSources)).Info("sources refreshed")
   357  	return nil
   358  }

   378  func (r *reaper) WriteMeasurements(ctx context.Context) {
   379  	var err error
   380  	for {
   381  		select {

   384  		case msg := <-r.measurementCh:
   385  			if err = r.SinksWriter.Write(msg); err != nil {
   386  				r.logger.Error(err)
   387  			}
   388  		}
