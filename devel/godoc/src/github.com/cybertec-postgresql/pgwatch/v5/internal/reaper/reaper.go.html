    88  // Reap() starts the main monitoring loop. It is responsible for fetching metrics measurements
    89  // from the sources and storing them to the sinks. It also manages the lifecycle of
    90  // the metric gatherers. In case of a source or metric definition change, it will
    91  // start or stop the gatherers accordingly.
    92  func (r *reaper) Reap(ctx context.Context) {
    93  	var err error
    94  	logger := r.logger
    95  
    96  	go r.WriteMeasurements(ctx)
    97  
    98  	r.ready.Store(true)
    99  
   100  	for { //main loop
   101  		if r.Logging.LogLevel == "debug" {
   102  			r.PrintMemStats()
   103  		}
   104  		if err = r.LoadSources(ctx); err != nil {
   105  			logger.WithError(err).Error("could not refresh active sources, using last valid cache")
   106  		}
   107  		if err = r.LoadMetrics(); err != nil {
   108  			logger.WithError(err).Error("could not refresh metric definitions, using last valid cache")
   109  		}
   110  
   111  		// UpdateMonitoredDBCache(r.monitoredSources)
   112  		hostsToShutDownDueToRoleChange := make(map[string]bool) // hosts went from master to standby and have "only if master" set
   113  		for _, monitoredSource := range r.monitoredSources {
   114  			src := monitoredSource.GetSource()
   115  			srcL := logger.WithField("source", src.Name)
   116  			ctx = log.WithLogger(ctx, srcL)
   117  
   118  			if monitoredSource.Connect(ctx, r.Sources) != nil {
   119  				r.WriteInstanceDown(src.Name)
   120  				srcL.Warning("could not init connection, retrying on next iteration")
   121  				continue
   122  			}
   123  
   124  			switch md := monitoredSource.(type) {
   125  			case *sources.DbConn:
   126  				if err = md.FetchRuntimeInfo(ctx, true); err != nil {
   127  					srcL.WithError(err).Error("could not start metric gathering")
   128  					continue
   129  				}
   130  
   131  				// Snapshot mutable RuntimeInfo fields under RLock so subsequent reads
   132  				// in this iteration are consistent and race-free against the per-source
   133  				// DbConnReaper goroutine calling FetchRuntimeInfo concurrently.
   134  				md.RLock()
   135  				isInRecovery := md.IsInRecovery
   136  				versionStr := md.VersionStr
   137  				DBSizeMB := md.ApproxDbSize / 1048576
   138  				var metricsConfig metrics.MetricIntervals
   139  				if md.IsInRecovery && len(md.MetricsStandby) > 0 {
   140  					metricsConfig = maps.Clone(md.MetricsStandby)
   141  				} else {
   142  					metricsConfig = maps.Clone(md.Metrics)
   143  				}
   144  				md.RUnlock()
   145  
   146  				srcL.WithField("recovery", isInRecovery).Infof("Connect OK. Version: %s", versionStr)
   147  				if isInRecovery && md.OnlyIfMaster {
   148  					srcL.Info("not added to monitoring due to 'master only' property")
   149  					if md.IsPostgresSource() {
   150  						srcL.Info("to be removed from monitoring due to 'master only' property and status change")
   151  						hostsToShutDownDueToRoleChange[src.Name] = true
   152  					}
   153  					continue
   154  				}
   155  
   156  				r.CreateSourceHelpers(ctx, srcL, md)
   157  
   158  				if md.IsPostgresSource() {
   159  					// only remove from monitoring when we're certain it's under the threshold
   160  					if DBSizeMB != 0 && DBSizeMB < r.Sources.MinDbSizeMB {
   161  						srcL.Infof("ignored due to the --min-db-size-mb filter, current size %d MB", DBSizeMB)
   162  						hostsToShutDownDueToRoleChange[src.Name] = true // for the case when DB size was previously above the threshold
   163  						continue
   164  					}
   165  
   166  					lastKnownStatusInRecovery := hostLastKnownStatusInRecovery[src.Name]
   167  					if lastKnownStatusInRecovery != md.IsInRecovery {
   168  						// metricsConfig was already selected above; here we only log the role change
   169  						if md.IsInRecovery && len(md.MetricsStandby) > 0 {
   170  							srcL.Warning("Switching metrics collection to standby config...")
   171  						} else if !md.IsInRecovery {
   172  							srcL.Warning("Switching metrics collection to primary config...")
   173  						}
   174  						// else: standby without a dedicated standby config keeps primary config, no warn
   175  					}
   176  				}
   177  				hostLastKnownStatusInRecovery[src.Name] = isInRecovery
   178  
   179  				// Sync metric names with sinks for the active config
   180  				for metricName := range metricsConfig {
   181  					mvp, metricDefExists := metricDefs.GetMetricDef(metricName)
   182  					if !metricDefExists {
   183  						epoch, ok := lastSQLFetchError.Load(metricName)
   184  						if !ok || ((time.Now().Unix() - epoch.(int64)) > 3600) {
   185  							srcL.WithField("metric", metricName).Warning("metric definition not found")
   186  							lastSQLFetchError.Store(metricName, time.Now().Unix())
   187  						}
   188  						continue
   189  					}
   190  					metricNameForStorage := metricName
   191  					if _, isSpecialMetric := specialMetrics[metricName]; !isSpecialMetric && mvp.StorageName > "" {
   192  						metricNameForStorage = mvp.StorageName
   193  					}
   194  					if err := r.SinksWriter.SyncMetric(src.Name, metricNameForStorage, sinks.AddOp); err != nil {
   195  						srcL.Error(err)
   196  					}
   197  				}
   198  
   199  				// Start SourceReaper for this source if not already running
   200  				if _, exists := r.cancelFuncs[src.Name]; !exists {
   201  					srcL.Info("starting source reaper")
   202  					sr := NewDbConnReaper(r, md)
   203  					sourceCtx, cancelFunc := context.WithCancel(ctx)
   204  					r.cancelFuncs[src.Name] = cancelFunc
   205  					go sr.Reap(sourceCtx)
   206  				}
   207  			case *sources.PromConn:
   208  				if _, exists := r.cancelFuncs[src.Name]; !exists {
   209  					srcL.Info("starting prometheus source reaper")
   210  					pr := NewPromSourceReaper(r, md)
   211  					sourceCtx, cancelFunc := context.WithCancel(ctx)
   212  					r.cancelFuncs[src.Name] = cancelFunc
   213  					go pr.Reap(sourceCtx)
   214  				}
   215  			}
   216  		}
   217  
   218  		r.ShutdownOldWorkers(ctx, hostsToShutDownDueToRoleChange)
   219  
   220  		r.prevLoopMonitoredDBs = slices.Clone(r.monitoredSources)
   221  		select {
   222  		case <-time.After(time.Second * time.Duration(r.Sources.Refresh)):
   223  			logger.Debugf("wake up after %d seconds", r.Sources.Refresh)
   224  		case <-ctx.Done():
   225  			return
   226  		}
   227  	}
   228  }

   230  // CreateSourceHelpers creates the extensions and metric helpers for the monitored source
   231  func (r *reaper) CreateSourceHelpers(ctx context.Context, srcL log.Logger, monitoredSource *sources.DbConn) {
   232  	if r.prevLoopMonitoredDBs.GetMonitoredDatabase(monitoredSource.Name) != nil {
   233  		return // already created
   234  	}
   235  	monitoredSource.RLock()
   236  	isInRecovery := monitoredSource.IsInRecovery
   237  	monitoredSource.RUnlock()
   238  	if !monitoredSource.IsPostgresSource() || isInRecovery {
   239  		return // no need to create anything for non-postgres sources
   240  	}
   241  
   242  	if r.Sources.TryCreateListedExtsIfMissing > "" {
   243  		srcL.Info("trying to create extensions if missing")
   244  		extsToCreate := strings.Split(r.Sources.TryCreateListedExtsIfMissing, ",")
   245  		extsCreated, err := monitoredSource.TryCreateMissingExtensions(ctx, extsToCreate)
   246  		if err != nil {
   247  			srcL.Warning(err)
   248  		}
   249  		if extsCreated != "" {
   250  			srcL.Infof("%d/%d extensions created: %s", len(extsCreated), len(extsToCreate), extsCreated)
   251  		}
   252  	}
   253  
   254  	if r.Sources.CreateHelpers {
   255  		srcL.Info("trying to create helper objects if missing")
   256  		if err := monitoredSource.TryCreateMetricsHelpers(ctx, func(metric string) string {
   257  			if m, ok := metricDefs.GetMetricDef(metric); ok {
   258  				return m.InitSQL
   259  			}
   260  			return ""
   261  		}); err != nil {
   262  			srcL.Warning(err)
   263  		}
   264  	}
   265  
   266  }

   268  func (r *reaper) ShutdownOldWorkers(ctx context.Context, hostsToShutDown map[string]bool) {
   269  	logger := r.logger
   270  	// loop over existing source reapers and stop if DB removed from config
   271  	// or state change makes it uninteresting
   272  	logger.Debug("checking if any workers need to be shut down...")
   273  	for sourceName, cancelFunc := range r.cancelFuncs {
   274  		var dbRemovedFromConfig bool
   275  
   276  		_, wholeDbShutDown := hostsToShutDown[sourceName]
   277  		if !wholeDbShutDown {
   278  			md := r.monitoredSources.GetMonitoredDatabase(sourceName)
   279  			if md == nil { // normal removing of DB from config
   280  				dbRemovedFromConfig = true
   281  				logger.Debugf("DB %s removed from config, shutting down source reaper...", sourceName)
   282  			}
   283  		}
   284  
   285  		if ctx.Err() != nil || wholeDbShutDown || dbRemovedFromConfig {
   286  			logger.WithField("source", sourceName).Info("stopping source reaper...")
   287  			cancelFunc()
   288  			delete(r.cancelFuncs, sourceName)
   289  			if err := r.SinksWriter.SyncMetric(sourceName, "", sinks.DeleteOp); err != nil {
   290  				logger.Error(err)
   291  			}
   292  		}
   293  	}
   294  
   295  	// Destroy conn pools and metric writers
   296  	r.CloseResourcesForRemovedMonitoredDBs(hostsToShutDown)
   297  }

   299  // LoadSources loads sources from the reader
   300  func (r *reaper) LoadSources(ctx context.Context) (err error) {
   301  	if DoesEmergencyTriggerfileExist(r.Metrics.EmergencyPauseTriggerfile) {
   302  		r.logger.Warningf("Emergency pause triggerfile detected at %s, ignoring currently configured DBs", r.Metrics.EmergencyPauseTriggerfile)
   303  		r.monitoredSources = make(sources.SourceConns, 0)
   304  		return nil
   305  	}
   306  
   307  	var newSrcs sources.SourceConns
   308  	srcs, err := r.SourcesReaderWriter.GetSources()
   309  	if err != nil {
   310  		return err
   311  	}
   312  	srcs = slices.DeleteFunc(srcs, func(s sources.Source) bool {
   313  		// filter out disabled sources and sources with group not in the list of groups to monitor
   314  		return !s.IsEnabled || len(r.Sources.Groups) > 0 && !slices.Contains(r.Sources.Groups, s.Group)
   315  	})
   316  
   317  	if newSrcs, err = srcs.ResolveDatabases(r.WriteInstanceDown); err != nil {
   318  		// discover dtabases for continuous monitoring sources
   319  		r.logger.WithError(err).Error("could not resolve databases from sources")
   320  	}
   321  
   322  	for i, newMD := range newSrcs {
   323  		md := r.monitoredSources.GetMonitoredDatabase(newMD.GetSource().Name)
   324  		if md == nil {
   325  			continue
   326  		}
   327  		if md.GetSource().Equal(newMD.GetSource()) {
   328  			// replace with the existing connection if the source is the same
   329  			newSrcs[i] = md
   330  			continue
   331  		}
   332  		// Source configs changed, stop all running gatherers to trigger a restart
   333  		// TODO: Optimize this for single metric addition/deletion/interval-change cases to not do a full restart
   334  		r.logger.WithField("source", md.GetSource().Name).Info("Source configs changed, restarting all gatherers...")
   335  		r.ShutdownOldWorkers(ctx, map[string]bool{md.GetSource().Name: true})
   336  	}
   337  	r.monitoredSources = newSrcs
   338  	r.logger.WithField("sources", len(r.monitoredSources)).Info("sources refreshed")
   339  	return nil
   340  }

   359  // WriteMeasurements() writes the metrics to the sinks
   360  func (r *reaper) WriteMeasurements(ctx context.Context) {
   361  	var err error
   362  	for {
   363  		select {
   364  		case <-ctx.Done():
   365  			return
   366  		case msg := <-r.measurementCh:
   367  			if err = r.SinksWriter.Write(msg); err != nil {
   368  				r.logger.Error(err)
   369  			}
   370  		}
   371  	}
   372  }
