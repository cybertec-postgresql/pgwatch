   120  func (pgw *PostgresWriter) init() (err error) {
   121  	return db.Init(pgw.ctx, pgw.sinkDb, func(ctx context.Context, conn db.PgxIface) error {
   122  		var isValidPartitionInterval bool
   123  		if err = conn.QueryRow(ctx,
   124  			"SELECT extract(epoch from $1::interval), extract(epoch from $2::interval), $3::interval >= '1h'::interval",
   125  			pgw.opts.RetentionInterval, pgw.opts.MaintenanceInterval, pgw.opts.PartitionInterval,
   126  		).Scan(&pgw.retentionInterval, &pgw.maintenanceInterval, &isValidPartitionInterval); err != nil {
   127  			return err
   128  		}
   129  
   130  		// epoch returns seconds but time.Duration represents nanoseconds
   131  		pgw.retentionInterval *= time.Second
   132  		pgw.maintenanceInterval *= time.Second
   133  
   134  		if !isValidPartitionInterval {
   135  			return fmt.Errorf("--partition-interval must be at least 1 hour, got: %s", pgw.opts.PartitionInterval)
   136  		}
   137  		if pgw.maintenanceInterval < 0 {
   138  			return errors.New("--maintenance-interval must be a positive PostgreSQL interval or 0 to disable it")
   139  		}
   140  		if pgw.retentionInterval < time.Hour && pgw.retentionInterval != 0 {
   141  			return errors.New("--retention must be at least 1 hour PostgreSQL interval or 0 to disable it")
   142  		}
   143  
   144  		exists, err := db.DoesSchemaExist(ctx, conn, "admin")
   145  		if err != nil || exists {
   146  			return err
   147  		}
   148  		for _, sql := range metricSchemaSQLs {
   149  			if _, err = conn.Exec(ctx, sql); err != nil {
   150  				return err
   151  			}
   152  		}
   153  		return nil
   154  	})
   155  }

   246  // poll is the main loop that reads from the input channel and flushes the data to the database
   247  func (pgw *PostgresWriter) poll() {
   248  	cache := make([]metrics.MeasurementEnvelope, 0, cacheLimit)
   249  	cacheTimeout := pgw.opts.BatchingDelay
   250  	tick := time.NewTicker(cacheTimeout)
   251  	for {
   252  		select {
   253  		case <-pgw.ctx.Done(): //check context with high priority
   254  			return
   255  		default:
   256  			select {
   257  			case entry := <-pgw.input:
   258  				cache = append(cache, entry)
   259  				if len(cache) < cacheLimit {
   260  					break
   261  				}
   262  				tick.Stop()
   263  				pgw.flush(cache)
   264  				cache = cache[:0]
   265  				tick = time.NewTicker(cacheTimeout)
   266  			case <-tick.C:
   267  				pgw.flush(cache)
   268  				cache = cache[:0]
   269  			case <-pgw.ctx.Done():
   270  				return
   271  			}
   272  		}
   273  	}
   274  }

   294  func (c *copyFromMeasurements) Next() bool {
   295  	for {
   296  		// Check if we need to advance to the next envelope
   297  		if c.envelopeIdx < 0 || c.measurementIdx+1 >= len(c.envelopes[c.envelopeIdx].Data) {
   298  			// Advance to next envelope
   299  			if ok := c.NextEnvelope(); !ok {
   300  				return false // No more envelopes
   301  			}
   302  			// Set metric name from first envelope, or detect metric boundary
   303  			if c.metricName == "" {
   304  				c.metricName = c.envelopes[c.envelopeIdx].MetricName
   305  			} else if c.metricName != c.envelopes[c.envelopeIdx].MetricName {
   306  				// We've hit a different metric - we're done with current metric
   307  				// Reset position to process this envelope on next call
   308  				c.envelopeIdx--
   309  				c.measurementIdx = len(c.envelopes[c.envelopeIdx].Data) // Set to length so we've "finished" this envelope
   310  				c.metricName = ""                                       // Reset for next metric
   311  				return false
   312  			}
   313  		}
   314  
   315  		// Advance to next measurement in current envelope
   316  		c.measurementIdx++
   317  		if c.measurementIdx < len(c.envelopes[c.envelopeIdx].Data) {
   318  			return true // Found valid measurement
   319  		}
   320  		// If we reach here, we've exhausted current envelope, loop will advance to next envelope
   321  	}
   322  }

   328  func (c *copyFromMeasurements) Values() ([]any, error) {
   329  	row := maps.Clone(c.envelopes[c.envelopeIdx].Data[c.measurementIdx])
   330  	tagRow := maps.Clone(c.envelopes[c.envelopeIdx].CustomTags)
   331  	if tagRow == nil {
   332  		tagRow = make(map[string]string)
   333  	}
   334  	for k, v := range row {
   335  		if after, ok := strings.CutPrefix(k, metrics.TagPrefix); ok {
   336  			tagRow[after] = fmt.Sprintf("%v", v)
   337  			delete(row, k)
   338  		}
   339  	}
   340  	jsonTags, terr := jsoniter.ConfigFastest.MarshalToString(tagRow)
   341  	json, err := jsoniter.ConfigFastest.MarshalToString(row)
   342  	if err != nil || terr != nil {
   343  		c.err = errors.Join(err, terr)
   344  		return nil, c.err
   345  	}
   346  	return []any{time.Unix(0, c.envelopes[c.envelopeIdx].Data.GetEpoch()), c.envelopes[c.envelopeIdx].DBName, json, jsonTags}, nil
   347  }

   361  // flush sends the cached measurements to the database
   362  func (pgw *PostgresWriter) flush(msgs []metrics.MeasurementEnvelope) {
   363  	if len(msgs) == 0 {
   364  		return
   365  	}
   366  	logger := log.GetLogger(pgw.ctx)
   367  	pgPartBounds := make(map[string]ExistingPartitionInfo) // metric=min/max
   368  	var err error
   369  
   370  	slices.SortFunc(msgs, func(a, b metrics.MeasurementEnvelope) int {
   371  		if a.MetricName < b.MetricName {
   372  			return -1
   373  		} else if a.MetricName > b.MetricName {
   374  			return 1
   375  		}
   376  		return 0
   377  	})
   378  
   379  	for _, msg := range msgs {
   380  		for _, dataRow := range msg.Data {
   381  			epochTime := time.Unix(0, metrics.Measurement(dataRow).GetEpoch())
   382  			bounds, ok := pgPartBounds[msg.MetricName]
   383  			if !ok || (ok && epochTime.Before(bounds.StartTime)) {
   384  				bounds.StartTime = epochTime
   385  				pgPartBounds[msg.MetricName] = bounds
   386  			}
   387  			if !ok || (ok && epochTime.After(bounds.EndTime)) {
   388  				bounds.EndTime = epochTime
   389  				pgPartBounds[msg.MetricName] = bounds
   390  			}
   391  		}
   392  	}
   393  
   394  	switch pgw.metricSchema {
   395  	case DbStorageSchemaPostgres:
   396  		err = pgw.EnsureMetricTimePartsExist(pgPartBounds)
   397  	case DbStorageSchemaTimescale:
   398  		err = pgw.EnsureMetricTimescale(pgPartBounds)
   399  	default:
   400  		logger.Fatal("unknown storage schema...")
   401  	}
   402  	pgw.forceRecreatePartitions = false
   403  	if err != nil {
   404  		select {
   405  		case pgw.lastError <- err:
   406  		default:
   407  		}
   408  	}
   409  
   410  	var rowsBatched, n int64
   411  	t1 := time.Now()
   412  	cfm := newCopyFromMeasurements(msgs)
   413  	for !cfm.EOF() {
   414  		n, err = pgw.sinkDb.CopyFrom(context.Background(), cfm.MetricName(), targetColumns[:], cfm)
   415  		rowsBatched += n
   416  		if err != nil {
   417  			logger.Error(err)
   418  			if _, ok := err.(*pgconn.ConnectError); ok {
   419  				logger.Errorf("Sink DB not reachable, dropping %d cached measurements", len(msgs))
   420  				break
   421  			}
   422  			if PgError, ok := err.(*pgconn.PgError); ok {
   423  				pgw.forceRecreatePartitions = PgError.Code == "23514"
   424  			}
   425  			if pgw.forceRecreatePartitions {
   426  				logger.Warning("Some metric partitions might have been removed, halting all metric storage. Trying to re-create all needed partitions on next run")
   427  			}
   428  		}
   429  	}
   430  	diff := time.Since(t1)
   431  	if err == nil {
   432  		logger.WithField("rows", rowsBatched).WithField("elapsed", diff).Info("measurements written")
   433  		return
   434  	}
   435  	select {
   436  	case pgw.lastError <- err:
   437  	default:
   438  	}
   439  }

   456  func (pgw *PostgresWriter) EnsureMetricTimePartsExist(metricPartBounds map[string]ExistingPartitionInfo) error {
   457  	var err error
   458  	var rows pgx.Rows
   459  	sqlEnsure := `select * from admin.ensure_partition_metric_time($1, $2, $3)`
   460  	for metric, pb := range metricPartBounds {
   461  		if pb.StartTime.IsZero() || pb.EndTime.IsZero() {
   462  			return fmt.Errorf("zero StartTime/EndTime in partitioning request: [%s:%v]", metric, pb)
   463  		}
   464  		partInfo, ok := pgw.partitionMapMetric[metric]
   465  		if !ok || pb.EndTime.After(partInfo.EndTime) || pb.EndTime.Equal(partInfo.EndTime) || pgw.forceRecreatePartitions {
   466  			if rows, err = pgw.sinkDb.Query(pgw.ctx, sqlEnsure, metric, pb.EndTime, pgw.opts.PartitionInterval); err != nil {
   467  				return err
   468  			}
   469  			if partInfo, err = pgx.CollectOneRow(rows, pgx.RowToStructByPos[ExistingPartitionInfo]); err != nil {
   470  				return err
   471  			}
   472  			pgw.partitionMapMetric[metric] = partInfo
   473  		}
   474  	}
   475  	return nil
   476  }

   602  		&migrator.MigrationNoTx{
   603  			Name: "01409 Switch to time-only partitioning",
   604  			Func: func(ctx context.Context, conn migrator.PgxIface) error {
   605  				const (
   606  					sqlDropOldEnsurePartitionDbnameTime = `DROP FUNCTION IF EXISTS admin.ensure_partition_metric_dbname_time;`
   607  
   608  					sqlListMetricTables = `SELECT c.relname
   609  						FROM pg_description d
   610  						JOIN pg_class c ON c.oid = d.objoid
   611  						WHERE d.description = 'pgwatch-generated-metric-lvl'`
   612  
   613  					sqlIsTableMigrated = `SELECT EXISTS (
   614  						SELECT 1 FROM pg_partitioned_table WHERE partrelid = to_regclass($1) AND partstrat = 'r')`
   615  
   616  					sqlRenameMetricTable = `ALTER TABLE %s RENAME TO %s`
   617  
   618  					sqlMetricTableBounds = `SELECT COALESCE(MIN(time), NOW()),
   619  						COALESCE(CEIL(EXTRACT(EPOCH FROM (MAX(time) - MIN(time))::interval) / 86400) + 1, 0)
   620  						FROM %s`
   621  
   622  					sqlEnsurePartitionMetricTime = `SELECT admin.ensure_partition_metric_time($1::text, $2::timestamptz, '1 day'::interval, $3)`
   623  
   624  					sqlListSubpartitions = `SELECT relid::regclass, parentrelid::regclass 
   625  						FROM pg_partition_tree(to_regclass($1)) WHERE relid::text LIKE 'subpartitions%' 
   626  						ORDER BY isleaf DESC, relid::text`
   627  
   628  					// moves rows into the new partitioned parent, detaches and drops the old subpartition;
   629  					// multiple statements in a single Exec run in an implicit transaction
   630  					sqlMoveSubpartition = `INSERT INTO %[1]s (time, dbname, data, tag_data) SELECT time, dbname, data, tag_data FROM %[2]s;
   631  						ALTER TABLE %[3]s DETACH PARTITION %[2]s;
   632  						DROP TABLE %[2]s;`
   633  
   634  					sqlDropTableIfExists = `DROP TABLE IF EXISTS %s`
   635  				)
   636  				if _, err := conn.Exec(ctx, sqlDropOldEnsurePartitionDbnameTime+
   637  					sqlMetricAdminFunctions+
   638  					sqlMetricEnsurePartitionPostgres); err != nil {
   639  					return err
   640  				}
   641  
   642  				rows, _ := conn.Query(ctx, sqlListMetricTables)
   643  				metricTables, err := pgx.CollectRows(rows, pgx.RowTo[string])
   644  				if err != nil {
   645  					return err
   646  				}
   647  
   648  				// skip *_before_v6_migration tables to avoid double migration
   649  				// this could happen if the migration is re-run after a failed attempt
   650  				const suffix = "_before_v6_migration"
   651  				for _, metricTableRawName := range metricTables {
   652  					if strings.HasSuffix(metricTableRawName, suffix) {
   653  						continue
   654  					}
   655  
   656  					metricTableRawRename := metricTableRawName + suffix
   657  					metricTableRenamed := pgx.Identifier{metricTableRawRename}.Sanitize()
   658  					metricTable := pgx.Identifier{metricTableRawName}.Sanitize()
   659  
   660  					// check if the table is already migrated to avoid errors on re-run after a failed migration attempt
   661  					var isTableMigrated bool
   662  					if err = conn.QueryRow(ctx, sqlIsTableMigrated, metricTableRawName).Scan(&isTableMigrated); err != nil {
   663  						return err
   664  					} else if isTableMigrated {
   665  						continue
   666  					}
   667  
   668  					err = pgx.BeginFunc(ctx, conn, func(tx pgx.Tx) (ferr error) {
   669  						if _, ferr = tx.Exec(ctx, fmt.Sprintf(sqlRenameMetricTable, metricTable, metricTableRenamed)); ferr != nil {
   670  							return
   671  						}
   672  
   673  						// for an empty table MIN(time) is NULL, so COALESCE falls back to server-side
   674  						// NOW() and daysToPrecreate becomes 0, creating a single empty partition
   675  						var minTime time.Time
   676  						var daysToPrecreate int32
   677  						if ferr = tx.QueryRow(ctx, fmt.Sprintf(sqlMetricTableBounds, metricTableRenamed)).Scan(&minTime, &daysToPrecreate); ferr == nil {
   678  							_, ferr = tx.Exec(ctx, sqlEnsurePartitionMetricTime, metricTableRawName, minTime, daysToPrecreate)
   679  						}
   680  						return
   681  					})
   682  
   683  					if err != nil {
   684  						return err
   685  					}
   686  
   687  					type partitionInfo struct {
   688  						Rel       string
   689  						ParentRel string
   690  					}
   691  					rows, _ := conn.Query(ctx, sqlListSubpartitions, metricTableRawRename)
   692  					partitionsInfo, err := pgx.CollectRows(rows, pgx.RowToStructByPos[partitionInfo])
   693  					if err != nil {
   694  						return err
   695  					}
   696  
   697  					for _, partInfo := range partitionsInfo {
   698  						if _, err := conn.Exec(ctx, fmt.Sprintf(sqlMoveSubpartition, metricTable, partInfo.Rel, partInfo.ParentRel)); err != nil {
   699  							return err
   700  						}
   701  					}
   702  
   703  					if _, err := conn.Exec(ctx, fmt.Sprintf(sqlDropTableIfExists, metricTableRenamed)); err != nil {
   704  						return err
   705  					}
   706  				}
   707  
   708  				return nil
   709  			},
   710  		},
